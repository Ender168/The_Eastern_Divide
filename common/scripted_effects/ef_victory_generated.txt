ef_setup_timed_victory = {
    if = {
        limit = {
            NOT = { has_global_flag = ef_timed_victory_initialized }
        }
        set_global_flag = ef_timed_victory_initialized
        set_variable = { global.ef_victory_days = 0 }
        set_variable = { global.ef_victory_next_drop = 105 }
        set_variable = { global.ef_fast_next_drop = 225 }
        set_variable = { global.ef_fast_threshold = 25 }
        set_variable = { global.ef_enemy_vp_threshold = 25 }
        ef_sample_victory_control = yes
    }
}

ef_activate_victory_missions = {
    WEF = { if = { limit = { check_variable = { var = global.ef_wef_enemy_vp value = global.ef_enemy_vp_threshold compare = greater_than_or_equals } NOT = { has_active_mission = ef_hold_standard } } activate_mission = ef_hold_standard } }
    WEF = { if = { limit = { check_variable = { var = global.ef_wef_enemy_vp value = global.ef_fast_threshold compare = greater_than_or_equals } NOT = { has_active_mission = ef_hold_fast } } activate_mission = ef_hold_fast } }
    EEF = { if = { limit = { check_variable = { var = global.ef_eef_enemy_vp value = global.ef_enemy_vp_threshold compare = greater_than_or_equals } NOT = { has_active_mission = ef_hold_standard } } activate_mission = ef_hold_standard } }
    EEF = { if = { limit = { check_variable = { var = global.ef_eef_enemy_vp value = global.ef_fast_threshold compare = greater_than_or_equals } NOT = { has_active_mission = ef_hold_fast } } activate_mission = ef_hold_fast } }
}

ef_daily_victory_clock = {
    ef_setup_timed_victory = yes
    add_to_variable = { global.ef_victory_days = 1 }
    if = { limit = { check_variable = { var = global.ef_victory_days value = global.ef_victory_next_drop compare = greater_than_or_equals } check_variable = { var = global.ef_enemy_vp_threshold value = 1 compare = greater_than } } subtract_from_variable = { global.ef_enemy_vp_threshold = 1 } add_to_variable = { global.ef_victory_next_drop = 105 } ef_sample_victory_control = yes ef_activate_victory_missions = yes }
    if = { limit = { check_variable = { var = global.ef_victory_days value = global.ef_fast_next_drop compare = greater_than_or_equals } check_variable = { var = global.ef_fast_threshold value = 1 compare = greater_than } } subtract_from_variable = { global.ef_fast_threshold = 1 } add_to_variable = { global.ef_fast_next_drop = 225 } ef_sample_victory_control = yes ef_activate_victory_missions = yes }
}

ef_monthly_victory = { ef_setup_timed_victory = yes ef_sample_victory_control = yes ef_activate_victory_missions = yes }

ef_standard_mission_expired = {
    ef_sample_victory_control = yes
    if = {
        limit = { tag = WEF check_variable = { var = global.ef_wef_enemy_vp value = global.ef_enemy_vp_threshold compare = greater_than_or_equals } }
        ef_register_victory_ready_wef = yes
    }
    else_if = {
        limit = { tag = EEF check_variable = { var = global.ef_eef_enemy_vp value = global.ef_enemy_vp_threshold compare = greater_than_or_equals } }
        ef_register_victory_ready_eef = yes
    }
}

ef_register_victory_ready_wef = {
    if = {
        limit = { NOT = { has_global_flag = ef_victory_resolution_queued } }
        set_global_flag = ef_victory_resolution_queued
        set_variable = { global.ef_victory_resolution_date = global.date }
        set_variable = { global.ef_victory_resolve_wef_vp = global.ef_wef_vp }
        set_variable = { global.ef_victory_resolve_eef_vp = global.ef_eef_vp }
        set_global_flag = ef_victory_ready_wef
        WEF = { country_event = { id = ef_victory.13 days = 1 } }
    }
    else_if = {
        limit = {
            check_variable = { var = global.ef_victory_resolution_date value = global.date compare = equals }
        }
        set_global_flag = ef_victory_ready_wef
    }
}

ef_register_victory_ready_eef = {
    if = {
        limit = { NOT = { has_global_flag = ef_victory_resolution_queued } }
        set_global_flag = ef_victory_resolution_queued
        set_variable = { global.ef_victory_resolution_date = global.date }
        set_variable = { global.ef_victory_resolve_wef_vp = global.ef_wef_vp }
        set_variable = { global.ef_victory_resolve_eef_vp = global.ef_eef_vp }
        set_global_flag = ef_victory_ready_eef
        WEF = { country_event = { id = ef_victory.13 days = 1 } }
    }
    else_if = {
        limit = {
            check_variable = { var = global.ef_victory_resolution_date value = global.date compare = equals }
        }
        set_global_flag = ef_victory_ready_eef
    }
}

ef_resolve_victory = {
    clr_global_flag = ef_victory_resolution_queued
    if = {
        limit = {
            has_global_flag = ef_victory_ready_wef
            has_global_flag = ef_victory_ready_eef
        }
        if = {
            limit = {
                check_variable = { var = global.ef_victory_resolve_wef_vp value = global.ef_victory_resolve_eef_vp compare = greater_than }
            }
            ef_finish_wef = yes
        }
        else_if = {
            limit = {
                check_variable = { var = global.ef_victory_resolve_eef_vp value = global.ef_victory_resolve_wef_vp compare = greater_than }
            }
            ef_finish_eef = yes
        }
        else = {
            ef_finish_draw = yes
        }
    }
    else_if = {
        limit = {
            has_global_flag = ef_victory_ready_wef
        }
        ef_finish_wef = yes
    }
    else_if = {
        limit = {
            has_global_flag = ef_victory_ready_eef
        }
        ef_finish_eef = yes
    }
    clr_global_flag = ef_victory_ready_wef
    clr_global_flag = ef_victory_ready_eef
}

ef_finish_wef = {
    set_global_flag = ef_match_finished
    set_global_flag = ef_result_wef
    WEF = { country_event = { id = ef_victory.1 } }
    EEF = { country_event = { id = ef_victory.1 } }
    WEF = { white_peace = EEF }
